# Flutter/Android Integration Guide for Vyamit Voice Assistant

## Overview
Your existing backend works perfectly with Flutter! You only need to:
1. Use LiveKit Flutter SDK instead of web SDK
2. Call your existing /api/token endpoint
3. Connect to the same LiveKit room

---

## 3. Basic Flutter Implementation

```dart
import 'package:flutter/material.dart';
import 'package:livekit_client/livekit_client.dart';
import 'package:http/http.dart' as http;
import 'dart:convert';
import 'package:permission_handler/permission_handler.dart';

class VoiceAssistantPage extends StatefulWidget {
  @override
  _VoiceAssistantPageState createState() => _VoiceAssistantPageState();
}

class _VoiceAssistantPageState extends State<VoiceAssistantPage> {
  Room? _room;
  bool _isConnected = false;
  bool _isConnecting = false;
  String _status = 'Disconnected';
  
  // Your backend URL
  final String backendUrl = 'http://YOUR_SERVER_IP:8000';
  
  @override
  void initState() {
    super.initState();
    _requestPermissions();
  }
  
  Future<void> _requestPermissions() async {
    await Permission.microphone.request();
  }
  
  // Get token from your existing backend
  Future<Map<String, String>> _getToken() async {
    final response = await http.post(
      Uri.parse('$backendUrl/api/token'),
      headers: {'Content-Type': 'application/json'},
      body: json.encode({
        'room_name': 'flutter-room-${DateTime.now().millisecondsSinceEpoch}',
        'participant_name': 'Mobile User',
      }),
    );
    
    if (response.statusCode == 200) {
      final data = json.decode(response.body);
      return {
        'server_url': data['server_url'],
        'participant_token': data['participant_token'],
      };
    }
    throw Exception('Failed to get token');
  }
  
  Future<void> _connect() async {
    if (_isConnecting) return;
    
    setState(() {
      _isConnecting = true;
      _status = 'Connecting...';
    });
    
    try {
      // Get token from your backend
      final tokenData = await _getToken();
      
      // Create room
      _room = Room();
      
      // Setup event listeners
      _room!.addListener(_onRoomUpdate);
      
      // Connect to LiveKit
      await _room!.connect(
        tokenData['server_url']!,
        tokenData['participant_token']!,
        roomOptions: const RoomOptions(
          defaultAudioCaptureOptions: AudioCaptureOptions(
            echoCancellation: true,
            noiseSuppression: true,
          ),
        ),
      );
      
      // Enable microphone
      await _room!.localParticipant?.setMicrophoneEnabled(true);
      
      setState(() {
        _isConnected = true;
        _isConnecting = false;
        _status = 'Connected - Listening';
      });
      
    } catch (e) {
      setState(() {
        _isConnecting = false;
        _status = 'Error: $e';
      });
    }
  }
  
  Future<void> _disconnect() async {
    await _room?.disconnect();
    await _room?.dispose();
    _room = null;
    
    setState(() {
      _isConnected = false;
      _status = 'Disconnected';
    });
  }
  
  void _onRoomUpdate() {
    setState(() {
      // Update UI based on room state
    });
  }
  
  @override
  void dispose() {
    _room?.dispose();
    super.dispose();
  }
  
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Vyamit Voice Assistant'),
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Icon(
              _isConnected ? Icons.mic : Icons.mic_off,
              size: 100,
              color: _isConnected ? Colors.green : Colors.grey,
            ),
            SizedBox(height: 20),
            Text(
              _status,
              style: TextStyle(fontSize: 18),
            ),
            SizedBox(height: 40),
            ElevatedButton(
              onPressed: _isConnecting 
                ? null 
                : _isConnected 
                  ? _disconnect 
                  : _connect,
              child: Text(
                _isConnecting 
                  ? 'Connecting...' 
                  : _isConnected 
                    ? 'Disconnect' 
                    : 'Connect',
              ),
            ),
          ],
        ),
      ),
    );
  }
}
```

## 10. API Service (Flutter)

```dart
// lib/services/api_service.dart
import 'package:http/http.dart' as http;
import 'dart:convert';

class ApiService {
  final String baseUrl;
  
  ApiService(this.baseUrl);
  
  Future<Map<String, dynamic>> getHealth() async {
    final response = await http.get(
      Uri.parse('$baseUrl/api/health'),
    );
    return json.decode(response.body);
  }
  
  Future<TokenResponse> getToken({
    String? roomName,
    String? participantName,
  }) async {
    final response = await http.post(
      Uri.parse('$baseUrl/api/token'),
      headers: {'Content-Type': 'application/json'},
      body: json.encode({
        if (roomName != null) 'room_name': roomName,
        if (participantName != null) 'participant_name': participantName,
      }),
    );
    
    if (response.statusCode == 200) {
      return TokenResponse.fromJson(json.decode(response.body));
    }
    throw Exception('Failed to get token: ${response.statusCode}');
  }
}

class TokenResponse {
  final String serverUrl;
  final String participantToken;
  
  TokenResponse({
    required this.serverUrl,
    required this.participantToken,
  });
  
  factory TokenResponse.fromJson(Map<String, dynamic> json) {
    return TokenResponse(
      serverUrl: json['server_url'],
      participantToken: json['participant_token'],
    );
  }
}
```

## 11. Run Commands

```bash
# Keep backend running
cd backend
.venv\Scripts\activate
uvicorn app.main:app --reload --host 0.0.0.0 --port 8000

# Keep agent worker running
cd backend
python -m app.agent.runner start

# Run Flutter app
cd flutter_app
flutter pub get
flutter run
```
